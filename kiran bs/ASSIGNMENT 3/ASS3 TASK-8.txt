#include <stdio.h>

struct Student {
    char name[50];
    int roll;
    float marks;
    char grade;
};

char calculateGrade(float marks) {
    if (marks >= 90)
        return 'A';
    else if (marks >= 75)
        return 'B';
    else if (marks >= 60)
        return 'C';
    else if (marks >= 40)
        return 'D';
    else
        return 'F';
}

int main() {
    int n;

    printf("Enter number of students: ");
    scanf("%d", &n);

    struct Student s[n];

    for (int i = 0; i < n; i++) {
        printf("\nEnter details of student %d\n", i + 1);

        printf("Name: ");
        scanf(" %[^\n]", s[i].name);

        printf("Roll Number: ");
        scanf("%d", &s[i].roll);

        printf("Marks: ");
        scanf("%f", &s[i].marks);

        s[i].grade = calculateGrade(s[i].marks);
    }

    printf("\n--- Student Records ---\n");
    for (int i = 0; i < n; i++) {
        printf("\nName: %s", s[i].name);
        printf("\nRoll Number: %d", s[i].roll);
        printf("\nMarks: %.2f", s[i].marks);
        printf("\nGrade: %c\n", s[i].grade);
    }

    return 0;
}
