#include <stdio.h>

int main() {
    FILE *fp;
    char text[100];

    /* ------------------ WRITE TO FILE ------------------ */
    fp = fopen("data.txt", "w");
    if (fp == NULL) {
        printf("Error opening file for writing.\n");
        return 1;
    }

    fprintf(fp, "Hello, this is the first line.\n");
    fprintf(fp, "C file handling is powerful.\n");

    fclose(fp);
    printf("Data written to file successfully.\n\n");

    /* ------------------ READ FROM FILE ------------------ */
    fp = fopen("data.txt", "r");
    if (fp == NULL) {
        printf("Error opening file for reading.\n");
        return 1;
    }

    printf("Reading from file:\n");
    while (fscanf(fp, " %[^\n]", text) != EOF) {
        printf("%s\n", text);
    }

    fclose(fp);

    /* ------------------ APPEND TO FILE ------------------ */
    fp = fopen("data.txt", "a");
    if (fp == NULL) {
        printf("Error opening file for appending.\n");
        return 1;
    }

    fprintf(fp, "This line was appended later.\n");
    fclose(fp);

    printf("\nData appended to file successfully.\n");

    return 0;
}
